using System;
using System.IO;
using UnityEngine;

namespace LittleRoad
{
    // Measurement only: replay actual applied inputs into independent rolling
    // predictions. No predicted state is returned to a driver or Rigidbody.
    // Known input replay isolates model error; it is NOT a future-input oracle
    // for prospective traffic or path prediction.
    public sealed class MotionPredictionAudit
    {
        const int Cars=24,Slots=16,Capacity=100000;
        struct Window
        {
            public bool active,flatContactFree;
            public int steps,contacts,recoveries,nextHorizon;
            public float start;
            public PlanarDrivePrediction state;
        }
        [Serializable] public struct Sample
        {
            public int id;public string equipment;
            public float start,end,horizon,positionError,speedError,headingError,yawError,steeringError;
            public float predictedBodyOffset,actualBodyOffset,startSpeed;
            public bool flatContactFree;
            public Vector3 predictedPosition,actualPosition;
        }
        [Serializable] sealed class Evidence
        {
            public string scope="Observation-only flat-tarmac model. Each window uses the real applied input history, not a held-input forecast. This measures physical model error, not future driving decisions or collision prediction. Contact/slope/off-surface windows remain recorded but separately labelled.";
            public string domain="Every step grounded, ground normal and body up y > 0.999, centre on tarmac, no new scored collision/recovery. Support contact effects remain in the measurements.";
            public string geometry="Actual transformed BoxCollider bottom corners versus predicted collider-centre-aware planar footprint; includes each body's measured half extents.";
            public float physicsStep;public int dropped,started;public int[] selectedIds;public Sample[] samples;
        }
        readonly bool[] enabled=new bool[Cars];readonly int[] tick=new int[Cars],slot=new int[Cars];
        readonly BoxCollider[] colliders=new BoxCollider[Cars];
        readonly Window[,] windows=new Window[Cars,Slots];readonly float[,] startSpeed=new float[Cars,Slots];
        readonly Sample[] samples=new Sample[Capacity];readonly int[] horizons;
        readonly int interval;readonly float dt;int count,dropped,started;
        public MotionPredictionAudit(string ids,float dt,int carCount)
        {
            if(carCount<1||carCount>Cars)throw new ArgumentException("Motion audit field capacity exceeded.");
            this.dt=dt;interval=Mathf.Max(1,Mathf.RoundToInt(.1f/dt));
            horizons=new[]{Mathf.RoundToInt(.1f/dt),Mathf.RoundToInt(.25f/dt),Mathf.RoundToInt(.5f/dt),Mathf.RoundToInt(1/dt)};
            if(ids=="all")for(int id=0;id<carCount;id++)enabled[id]=true;
            else foreach(string value in ids.Split(',')) {
                if(!int.TryParse(value,out int id)||id<0||id>=carCount)throw new ArgumentException("Motion audit requires IDs present in this field, or all.");
                enabled[id]=true;
            }
        }
        public void Observe(RoadCar car,float time)
        {
            int id=car.driverId;if(id<0||id>=Cars||!enabled[id]||car.finished)return;
            bool domain=(car.ground==null||car.ground.grounded&&car.ground.normal.y>.999f)&&car.transform.up.y>.999f&&Mathf.Abs(car.offset)<Circuit.Width*.5f;
            var shape=colliders[id];if(shape==null)colliders[id]=shape=car.GetComponent<BoxCollider>();
            var actual=new PlanarDrivePrediction(car.body.position,car.body.linearVelocity,car.transform.forward,car.body.angularVelocity.y,car.SteeringState);
            for(int j=0;j<Slots;j++) {
                ref var w=ref windows[id,j];if(!w.active)continue;
                w.flatContactFree&=domain&&car.contacts==w.contacts&&car.recoveries==w.recoveries;
                if(w.steps==horizons[w.nextHorizon]) {
                    var delta=w.state.position-actual.position;delta.y=0;
                    if(count<Capacity)samples[count++]=new Sample{id=id,equipment=car.equipment,start=w.start,end=time,horizon=w.steps*dt,
                        positionError=delta.magnitude,speedError=Mathf.Abs(w.state.velocity.magnitude-actual.velocity.magnitude),
                        headingError=Vector3.Angle(w.state.forward,actual.forward),yawError=Mathf.Abs(w.state.yawRate-actual.yawRate),
                        steeringError=Mathf.Abs(w.state.steering-actual.steering),flatContactFree=w.flatContactFree,startSpeed=startSpeed[id,j],
                        predictedBodyOffset=w.state.MaxBodyOffset(car.race.track,shape.center,shape.size*.5f),
                        actualBodyOffset=ActualBodyOffset(car,shape),
                        predictedPosition=w.state.position,actualPosition=actual.position};else dropped++;
                    if(++w.nextHorizon==horizons.Length){w.active=false;continue;}
                }
            }
            if(tick[id]++%interval==0) {
                int j=slot[id]++%Slots;
                if(windows[id,j].active)throw new InvalidOperationException("Motion audit window capacity exceeded.");
                windows[id,j]=new Window{active=true,flatContactFree=domain,state=actual,start=time,contacts=car.contacts,recoveries=car.recoveries};
                startSpeed[id,j]=car.speed;started++;
            }
            // Inputs for THIS physical step are known now. Compare their result
            // with next tick's observation before consuming any later input.
            for(int j=0;j<Slots;j++) {
                ref var w=ref windows[id,j];if(!w.active)continue;
                w.state.Step(car.tuning,car.throttle,car.steering,dt);w.steps++;
            }
        }
        static float ActualBodyOffset(RoadCar car,BoxCollider shape)
        {
            float maximum=0;var half=shape.size*.5f;
            for(int side=-1;side<=1;side+=2)for(int end=-1;end<=1;end+=2) {
                var point=car.transform.TransformPoint(shape.center+Vector3.Scale(half,new Vector3(side,-1,end)));
                car.race.track.Nearest(point,out float offset);maximum=Mathf.Max(maximum,Mathf.Abs(offset));
            }
            return maximum;
        }
        public void Save(string report)
        {
            var copy=new Sample[count];Array.Copy(samples,copy,count);
            var ids=new System.Collections.Generic.List<int>();for(int i=0;i<Cars;i++)if(enabled[i])ids.Add(i);
            var evidence=new Evidence{physicsStep=dt,dropped=dropped,started=started,selectedIds=ids.ToArray(),samples=copy};
            File.WriteAllText(Path.ChangeExtension(report,"motion-prediction.json"),JsonUtility.ToJson(evidence,true));
        }
    }
}
